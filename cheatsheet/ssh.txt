## Connecting to tfpool

Add this config to .ssh/config:

```
Host tfpool*
  HostName %h
  User velikanm
  ProxyJump velikanm@login.informatik.uni-freiburg.de
  IdentityFile ~/.ssh/id_rsa

Host poolgpu*
  HostName %h
  User velikanm
  ProxyJump velikanm@login.informatik.uni-freiburg.de
  IdentityFile ~/.ssh/id_rsa

# enumerate imbip-compute-*
Host tfpool42
Host tfpool64
Host poolgpu2
```

Connect to remote:
$ ssh velikanm@login.informatik.uni-freiburg.de

Connect to one of GPUs:
$ ssh tfpool42 # 32, 34, 38, 41, 42
$ ssh poolgpu2

# Range of host numbers
START=0
END=99

# Timeout settings
SSH_TIMEOUT=1

# Loop through tfpool00 to tfpool99
for i in $(seq -w $START $END); do
HOST="tfpool$i"

echo "Checking $HOST..."

# Check SSH connectivity and run nvidia-smi to check for GPUs
ssh -o ConnectTimeout=$SSH_TIMEOUT -o BatchMode=yes -o StrictHostKeyChecking=no "${HOST}" "nvidia-smi --query-gpu=name --format=csv,noheader" 2>/dev/null | grep -q .

if [ $? -eq 0 ]; then
echo "✅ $HOST: reachable and GPU available."
else
echo "❌ $HOST: not reachable or no GPU detected."
fi
done
